## Priority 4: Code Quality

### Fix 4.1: Unify error handling in storage

**File:** `src/utils/storage.ts`
**Problem:** Inconsistent try/catch — some log errors, some are silent.
**Action:** Create a safe wrapper:

```tsx
function safeGet<T>(key: string, fallback: T): T {
  try {
    const raw = localStorage.getItem(key);
    if (raw) return JSON.parse(raw) as T;
  } catch (e) {
    console.error(`Error reading ${key}:`, e);
  }
  return fallback;
}

function safeSet(key: string, value: unknown): void {
  try {
    localStorage.setItem(key, JSON.stringify(value));
  } catch (e) {
    console.error(`Error writing ${key}:`, e);
  }
}
```

Replace all raw localStorage calls with these helpers.

### Fix 4.2: Optimize `getPantryItemName` and `getPantryItemEmoji`

**File:** `src/utils/storage.ts` lines 1021-1032
**Problem:** These call `getAllPantryItems()` on every invocation — O(n) per call, called in loops.
**Action:** Create a cached Map:

```tsx
let pantryCache: Map<string, PantryItem> | null = null;

function getPantryCache(): Map<string, PantryItem> {
  if (!pantryCache) {
    pantryCache = new Map(getAllPantryItems().map((p) => [p.id, p]));
  }
  return pantryCache;
}

// Invalidate cache when pantry changes:
export function invalidatePantryCache() {
  pantryCache = null;
}
```

Then use the cache in `getPantryItemName`, `getPantryItemEmoji`, and `getUnifiedCardDataset`.

### Fix 4.3: Add Error Boundary

**File:** Create `src/components/ErrorBoundary.tsx`
**Action:** Create a class-based error boundary wrapping the app:

```tsx
// src/components/ErrorBoundary.tsx
import React from "react";

interface Props {
  children: React.ReactNode;
}
interface State {
  hasError: boolean;
  error?: Error;
}

export class ErrorBoundary extends React.Component<Props, State> {
  state: State = { hasError: false };

  static getDerivedStateFromError(error: Error) {
    return { hasError: true, error };
  }

  render() {
    if (this.state.hasError) {
      return (
        <div className="flex flex-col items-center justify-center h-screen p-8 text-center">
          <span className="text-4xl mb-4">😵</span>
          <h2 className="text-lg font-bold mb-2">Algo salió mal</h2>
          <p className="text-sm text-zinc-500 mb-4">
            {this.state.error?.message}
          </p>
          <button
            onClick={() => window.location.reload()}
            className="px-4 py-2 rounded-full bg-zinc-900 text-white text-xs font-semibold"
          >
            Recargar
          </button>
        </div>
      );
    }
    return this.props.children;
  }
}
```

Wrap in `main.tsx`:

```tsx
<StrictMode>
  <ErrorBoundary>
    <App />
  </ErrorBoundary>
</StrictMode>
```
